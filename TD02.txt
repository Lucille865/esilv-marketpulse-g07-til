>>> import json
>>> import csv

>>>
>>> DATA_DIR = Path("data/sample")

>>> # 1. Chargement des données
>>> with open(DATA_DIR / "instruments.json", encoding="utf-8") as file:
...     instruments = json.load(file)
...
>>> with open(DATA_DIR / "prices.csv", encoding="utf-8") as file:
...     prices = list(csv.DictReader(file))
...
>>> # Fonctions utilitaires
>>> def filter_prices(prices, ticker):
...     return [row for row in prices if row["ticker"] == ticker]
...
>>> def get_first_close(prices):
...     return float(prices[0]["close"])
...
>>> def get_last_close(prices):
...     return float(prices[-1]["close"])
...
>>> # 2. Fonction réutilisable pour afficher le résumé de marché
>>> def display_market_summary(asset, prices, show_currency=True):
...     ticker = asset["ticker"]
...     name = asset["name"]
...     nb_obs = len(prices)
...     first = get_first_close(prices)
...     last = get_last_close(prices)
...
...     currency_suffix = f" {asset['currency']}" if show_currency and "currency" in asset else ""
...
...     print(f"{ticker} - {name}")
...     print(f"Observations : {nb_obs}")
...     print(f"First close  : {first:.2f}{currency_suffix}")
...     print(f"Last close   : {last:.2f}{currency_suffix}")
...     print()
...

>>> benchmark = instruments["benchmark"]
